{
    app.MapOpenApi(); // [EXISTENTE si ya estaba]
}

// [PASO 2] UseAuthentication → UsuarioActualMiddleware → UseAuthorization

// [EXISTENTE] Endpoint /api/health/db (mantener su implementación original).

// [EXISTENTE] Si usa controllers: app.MapControllers();

// [PASO 2] Endpoints nuevos
